# 点云工具 · 高精度模式后端依赖
#
# 不要直接 pip install -r 这个文件 —— torch 必须从 PyTorch 自己的源装 CUDA 版本，
# 请用 setup.ps1，它会按正确顺序和正确的源来装。
#
# ---- 装法（setup.ps1 实际执行的）----
#
# 1) PyTorch CUDA 版（约 2.5GB）
#    本机驱动 546.33 = CUDA 12.3，选 cu121 是确定兼容的（cu126 需依赖小版本兼容，不赌）
#    pip install torch==2.5.1 torchvision --index-url https://download.pytorch.org/whl/cu121
#
# 2) MoGe 及其依赖（会连带装 utils3d / gradio 等）
#    pip install git+https://github.com/microsoft/MoGe.git
#
# 3) Web 服务
#    pip install fastapi "uvicorn[standard]" python-multipart
#
# ---- 运行时实际用到的 ----
torch>=2.0.0          # 从 cu121 源装
torchvision           # 同上
moge                  # 从 GitHub 装
fastapi
uvicorn[standard]
python-multipart      # FastAPI 处理 multipart/form-data 上传必需，容易漏
pillow
numpy
opencv-python
huggingface_hub

# ---- 生成式 3D 模式（TripoSR，源码 vendor 在 tsr/）----
# torchmcubes 被刻意排除：git+ 源码依赖且要编译 C++，Windows 必坑，
# tsr/models/isosurface.py 已改用 scikit-image 的 marching_cubes
omegaconf
einops
# 上下界都是硬约束：4.35（TripoSR 上游锁定）的 tokenizers 会把 huggingface_hub
# 拽回 0.17 弄坏 MoGe；5.x 重构了 ViT 参数命名，TripoSR 的 checkpoint 加载不上；
# hy3dgen 又要求 >=4.48。只有 4.48–4.x 这段同时满足三方。
transformers>=4.48,<5
trimesh
rembg                 # 抠背景；u2net.onnx 缓存位置由 U2NET_HOME 控制（start.ps1 指到项目盘）
onnxruntime           # rembg 的推理运行时（CPU 版即可）
scikit-image
imageio               # tsr.utils 顶层 import，实际只有导视频用到

# ---- 多视图生成式（Hunyuan3D-2，仅形状管线）----

diffusers
accelerate

pygltflib
